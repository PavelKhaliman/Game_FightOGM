package render

import (
	"fightogm/internal/assets"
	"fightogm/internal/combat"
	"fightogm/internal/fighter"
	"fightogm/internal/game"
	"fmt"
	rl "github.com/gen2brain/raylib-go/raylib"
	"math"
	"path/filepath"
)

type Renderer struct {
	Root      string
	Manifest  assets.Manifest
	Shader    rl.Shader
	Models    []*assets.Model
	Portraits []rl.RenderTexture2D
	Actors    [2]*assets.Model
	Camera    VersusCamera
	Effects   Effects
	Time      float32
	Sign      rl.RenderTexture2D
}

func New(root string, roster []fighter.Definition, manifest assets.Manifest, font rl.Font, progress func(int, string)) *Renderer {
	r := &Renderer{Root: root, Manifest: manifest, Shader: LoadLighting(), Camera: NewCamera()}
	r.Sign = rl.LoadRenderTexture(1024, 128)
	rl.BeginTextureMode(r.Sign)
	rl.ClearBackground(rl.NewColor(35, 45, 53, 255))
	rl.DrawRectangleLines(2, 2, 1020, 124, rl.NewColor(211, 162, 71, 255))
	rl.DrawTextEx(font, "ОГМ  /  ПРОИЗВОДСТВО  /  РАБОТАЕМ", rl.NewVector2(40, 44), 40, 1, rl.NewColor(240, 205, 143, 255))
	rl.EndTextureMode()
	for i, d := range roster {
		progress(i, d.Name)
		m := assets.Load(filepath.Join(root, d.ModelPath), manifest.Characters[d.ID], r.Shader)
		r.Models = append(r.Models, m)
		var portrait rl.RenderTexture2D
		if m.Available {
			portrait = rl.LoadRenderTexture(256, 256)
			rl.BeginTextureMode(portrait)
			rl.ClearBackground(rl.NewColor(23, 32, 44, 255))
			camera := rl.Camera3D{Position: rl.NewVector3(.5, 1.5, 3.25), Target: rl.NewVector3(0, 1.12, 0), Up: rl.NewVector3(0, 1, 0), Fovy: 35, Projection: rl.CameraPerspective}
			rl.BeginMode3D(camera)
			DrawShadow(rl.Vector3{}, 1)
			rl.DrawModelEx(m.Model, rl.Vector3{}, rl.NewVector3(0, 1, 0), -7, rl.NewVector3(m.Scale, m.Scale, m.Scale), rl.White)
			rl.EndMode3D()
			rl.EndTextureMode()
		}
		r.Portraits = append(r.Portraits, portrait)
	}
	return r
}
func (r *Renderer) SetMatch(m *game.Match) error {
	for i, actor := range r.Actors {
		if actor != nil {
			actor.Close()
			r.Actors[i] = nil
		}
	}
	for i, f := range m.Fighters {
		model := assets.Load(filepath.Join(r.Root, f.Definition.ModelPath), r.Manifest.Characters[f.Definition.ID], r.Shader)
		r.Actors[i] = model
		if !model.Available {
			return fmt.Errorf("%s: %s", f.Definition.Name, model.Error)
		}
	}
	r.Camera = NewCamera()
	r.Effects = Effects{}
	return nil
}
func (r *Renderer) Events(events []game.Event) {
	for _, e := range events {
		r.Effects.Spawn(e)
		if e.Kind == "hit" || e.Kind == "heavy" || e.Kind == "kick" || e.Kind == "ko" || e.Kind == "counter" {
			r.Camera.Shake = max(r.Camera.Shake, e.Strength)
		}
		if e.Kind == "ultimate" {
			r.Camera.Cinematic = 1.6
			r.Camera.Actor = e.Player
		}
	}
}
func (r *Renderer) Update(m *game.Match, dt, animationDT float32) {
	r.Time += dt
	r.Effects.Update(animationDT)
	if m == nil {
		return
	}
	r.Camera.Update(m, dt)
	for i, f := range m.Fighters {
		model := r.Actors[i]
		if model == nil || !model.Available {
			continue
		}
		model.Controller.Play(model, f.Animation, f.AnimationLoop, f.AnimationDuration, f.AnimationSerial)
		model.Controller.Update(model, animationDT)
	}
}
func (r *Renderer) DrawBattle(m *game.Match, debug bool) {
	rl.BeginMode3D(r.Camera.View())
	DrawArena(r.Time)
	r.drawSign(r.Camera.View())
	for i, f := range m.Fighters {
		model := r.Actors[i]
		if model == nil || !model.Available {
			continue
		}
		pos := vector(f.Position)
		DrawShadow(pos, 1)
		tint := rl.White
		if f.Status.Omni > 0 {
			tint = rl.NewColor(255, 161, 143, 255)
			DrawAura(pos, r.Time, "omni")
		}
		if f.Status.Gym > 0 {
			DrawAura(pos, r.Time, "gym")
		}
		if f.Status.Focus > 0 {
			DrawAura(pos, r.Time, "focus")
		}
		if f.Status.Armor > 0 {
			DrawAura(pos, r.Time, "armor")
		}
		if f.Status.Invisible > 0 {
			tint = rl.Fade(tint, .07)
		}
		yaw := float32(math.Atan2(float64(f.Facing.X), float64(f.Facing.Z)) * 180 / math.Pi)
		scale := rl.NewVector3(model.Scale, model.Scale, model.Scale)
		if f.Move != nil && f.Move.Effect == "afterimage" {
			for j := 3; j > 0; j-- {
				p := vector(f.Position.Sub(f.Facing.Mul(float32(j) * .22)))
				rl.DrawModelEx(model.Model, p, rl.NewVector3(0, 1, 0), yaw, scale, rl.Fade(EffectColor("afterimage"), .1*float32(4-j)))
			}
		}
		rl.DrawModelEx(model.Model, pos, rl.NewVector3(0, 1, 0), yaw, scale, tint)
		marker := rl.NewColor(245, 105, 82, 255)
		if i == 1 {
			marker = rl.NewColor(78, 191, 225, 255)
		}
		rl.DrawCircle3D(rl.NewVector3(pos.X, .021, pos.Z), .43, rl.NewVector3(1, 0, 0), 90, rl.Fade(marker, .75))
		if f.Move != nil && !f.Move.Projectile {
			p := f.Position.Add(f.Facing.Mul(.55))
			p.Y += 1.1
			s := float32(1)
			if f.Move.Ultimate {
				s = 1.6
			}
			DrawProp(f.Move.Effect, p, f.Facing, r.Time, s)
		}
		if f.Move != nil && f.Move.Effect == "panel" {
			DrawProp("panel", f.Position.Add(f.Facing.Mul(.7)).Add(combat.Vec3{Y: 1.5}), f.Facing, r.Time, 1)
		}
		if debug {
			for _, h := range f.Hurtboxes() {
				rl.DrawSphereWires(vector(h.Center), h.Radius, 6, 10, rl.Green)
			}
			for _, h := range f.Hitboxes() {
				rl.DrawSphereWires(vector(h.Center), h.Radius, 6, 10, rl.Red)
			}
		}
	}
	for _, p := range m.Projectiles {
		if p.Delay <= 0 {
			DrawProp(p.Kind, p.Position, p.Direction, r.Time, .6)
		}
	}
	r.Effects.Draw()
	rl.EndMode3D()
}
func (r *Renderer) DrawShowcase(selected [2]int, menu bool) {
	camera := rl.Camera3D{Position: rl.NewVector3(0, 2.8, 7.6), Target: rl.NewVector3(0, 1.2, 0), Up: rl.NewVector3(0, 1, 0), Fovy: 36, Projection: rl.CameraPerspective}
	if menu {
		camera.Position.X = -1
		camera.Target.X = -1
	}
	rl.BeginMode3D(camera)
	DrawArena(r.Time)
	r.drawSign(camera)
	for p, i := range selected {
		if i < 0 || i >= len(r.Models) {
			continue
		}
		model := r.Models[i]
		if !model.Available {
			continue
		}
		x := float32(-1.55 + float32(p)*3.1)
		if menu {
			x = float32(.55 + float32(p)*1.55)
		}
		pos := rl.NewVector3(x, 0, 0)
		DrawShadow(pos, 1)
		yaw := float32(25)
		if p == 1 {
			yaw = -25
		}
		rl.DrawModelEx(model.Model, pos, rl.NewVector3(0, 1, 0), yaw, rl.NewVector3(model.Scale, model.Scale, model.Scale), rl.White)
	}
	rl.EndMode3D()
}
func (r *Renderer) UpdateShowcase(dt float32, selected [2]int) {
	r.Time += dt
	for p, i := range selected {
		if p == 1 && selected[0] == i {
			continue
		}
		if i >= 0 && i < len(r.Models) && r.Models[i].Available {
			r.Models[i].Controller.Update(r.Models[i], dt)
		}
	}
}
func (r *Renderer) Close() {
	for _, m := range r.Actors {
		if m != nil {
			m.Close()
		}
	}
	for _, m := range r.Models {
		m.Close()
	}
	for _, t := range r.Portraits {
		if t.ID > 0 {
			rl.UnloadRenderTexture(t)
		}
	}
	rl.UnloadShader(r.Shader)
	rl.UnloadRenderTexture(r.Sign)
}

func (r *Renderer) drawSign(camera rl.Camera3D) {
	rl.DrawBillboardRec(camera, r.Sign.Texture, rl.NewRectangle(0, 0, 1024, -128), rl.NewVector3(0, 3.05, -5.7), rl.NewVector2(5.8, .72), rl.White)
}
